cmake_minimum_required(VERSION 3.22)
project(Ironfist VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(Ironfist COMPANY_NAME "Bob Audio" PLUGIN_MANUFACTURER_CODE BobA PLUGIN_CODE Irnf FORMATS VST3 PRODUCT_NAME "IRONFIST" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(Ironfist PRIVATE Source.cpp)
target_compile_definitions(Ironfist PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Ironfist PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
